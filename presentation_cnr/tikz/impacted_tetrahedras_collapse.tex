			\begin{tikzpicture}[tdplot_main_coords, scale=2]

				% Vertices
				\coordinate (A) at (0,0,0);
				\coordinate (B) at (1,0,0);
				\coordinate (C) at (0.5,0.866,0); % equilateral basis
				\coordinate (D) at (0.5,0.2887,0.8165);
				\coordinate (E) at (1.4,0.4,0.55);
				\coordinate (F) at (-0.4,0.4,0.55);
				\onslide<1>{
					\draw[fill = red!20] (A) -- (B) -- (D);
					\draw[fill = red!10] (B) -- (C) -- (D);
				}

				\onslide<2>{
					\draw[fill = blue!20] (B) -- (D) -- (E);
					\draw[fill = blue!30] (B) -- (C) -- (D);
				}
				\onslide<3->{
					\draw[fill = red!20] (A) -- (B) -- (D);
					\draw[fill = blue!20] (B) -- (D) -- (E);
					\draw[fill = violet!30] (B) -- (C) -- (D);
				}

				% Edges
				\draw[thick, line cap=round] (A) -- (D);
				\draw[thick, line cap=round] (A) -- (F);
				\draw[thick, line cap=round] (D) -- (F);
				\draw[thick, line cap=round] (B) -- (D);
				\draw[dashed, line cap=round] (C) -- (D);

				\draw[dashed, line cap=round] (B) -- (C);
				\draw[thick, line cap=round] (B) -- (E);

				\draw[dashed, line cap=round] (C) -- (F);
				\draw[thick, line cap=round] (D) -- (E);
				\draw[dashed, line cap=round] (C) -- (A);

				% Points
				\draw (A) node[anchor=north east] {A};
				\draw (B) node[anchor=north east] {B};
				\draw (C) node[anchor=north west] {C};
				\draw (D) node[anchor=south west] {D};
				\draw (E) node[anchor=south west] {E};
				\draw (F) node[anchor=south east] {F};

			\end{tikzpicture}
		};

		\node (After) at (6,0) {
			% Edge collapse after
			\tdplotsetmaincoords{70}{50} % view angle
			\begin{tikzpicture}[tdplot_main_coords, scale=2]

				% Vertices
				\coordinate (A) at (0,0,0);
				\coordinate (C) at (0.5,0.866,0); % equilateral basis
				\coordinate (D) at (0.5,0.2887,0.8165);
				\coordinate (E) at (1.4,0.4,0.55);
				\coordinate (F) at (-0.4,0.4,0.55);

				\onslide<2->{
					\draw[fill = green!20] (A) -- (C) -- (E);

				\draw[thick, line cap=round] (D) -- (F);

				\draw[dashed, line cap=round] (C) -- (F);
				\draw[thick, line cap=round] (D) -- (E);
				\draw[thick, line cap=round] (C) -- (A);

				% Points

				\draw (C) node[anchor=north west] {C};
				\draw (D) node[anchor=south west] {D};
				\draw (E) node[anchor=south west] {E};
				\draw (F) node[anchor=south east] {F};

			\end{tikzpicture}
		};


		% Arrow between them
		\draw[->, thick] (Before.east) -- (After.west);

	\end{tikzpicture}
