\begin{frame}
	\frametitle{Topological failure}

	\onslide<1->{Collapsing $e:= (u,v)$ can break topology}

	\onslide<2->{
		\begin{equation}
			\tag{Link condition}
			\Lk{e} = \Lk{u} \cap \Lk{v}
			\label{eqn:linkCondition}
		\end{equation}


		\begin{itemize}
			\item $\Lk{u}$ = neighbours of $u$
			\item $\Lk{e}$ = vertices sharing a face with $e$
		\end{itemize}
