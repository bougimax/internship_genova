    % First tetrahedron (Before)
    \node (Before) at (0,0) {
        % Face swap before
        \tdplotsetmaincoords{55}{30} % view angle
        \begin{tikzpicture}[tdplot_main_coords, scale=3]

          % Vertices
          \coordinate (A2) at (0.1,-0.5,0);
          \coordinate (A1) at (-0.1,-0.5,0);
          \coordinate (B2) at (0.1,0.5,0);
          \coordinate (B1) at (-0.1,0.5,0);
          \coordinate (C2) at (0.1,0,1);
          \coordinate (C1) at (-0.1,0,1);
          \coordinate (D1) at (-1,0,0.5);
          \coordinate (D2) at (1,0,0.5);

          \draw[thick, rounded corners=1pt, fill=red!30] (A2) -- (B2) -- (C2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (A2) -- (B2) -- (D2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (D2) -- (B2) -- (C2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (A2) -- (D2) -- (C2) -- cycle;

          % Edges
          \draw[dashed, line cap=round] (D1) -- (B1);
          \draw[dashed, line cap=round] (C1) -- (B1);
          \draw[dashed, line cap=round] (A1) -- (B1);
          \draw[dashed, line cap=round] (A2) -- (B2);
          \draw[dashed, line cap=round] (C2) -- (B2);
          \draw[dashed, line cap=round] (D2) -- (B2);

          % % Points

          \draw (B2) node[anchor=south east] {B};
          \draw (C1) node[anchor=south west] {C};
          \draw (D1) node[anchor=south west] {D};
          \draw (D2) node[anchor=south west] {E};

        \end{tikzpicture}
    };

    % Second tetrahedron (After)
    \node (After) at (7.5,0) {
        % Face swap after
        \tdplotsetmaincoords{55}{30} % view angle
        \begin{tikzpicture}[tdplot_main_coords, scale=3]

          % Vertices
          \coordinate (A1) at (0,-0.5,-0.05);
          \coordinate (A3) at (0,-0.5,0.05);
          \coordinate (B1) at (0,0.5,-0.05);
          \coordinate (B2) at (0,0.5,0.05);
          \coordinate (C2) at (0,0.05,1);
          \coordinate (C3) at (0,-0.05,1);
          \coordinate (D3) at (-1,-0.1,0.45);
          \coordinate (D2) at (-1,0.1,0.45);
          \coordinate (D1) at (-1,0,0.25);
          \coordinate (E3) at (1,-0.1,0.45);
          \coordinate (E2) at (1,0.1,0.45);
          \coordinate (E1) at (1,0,0.25);

          % Faces
          \draw[thick, rounded corners=1pt, fill=blue!30] (A1) -- (B1) -- (E1) -- cycle;
          \draw[thick, rounded corners=1pt, fill=blue!30] (A1) -- (D1) -- (B1) -- cycle;
          \draw[thick, rounded corners=1pt, fill=blue!30] (A1) -- (D1) -- (E1) -- cycle;
          \draw[thick, rounded corners=1pt, fill=blue!30] (B1) -- (D1) -- (E1) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (C2) -- (B2) -- (E2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (C2) -- (D2) -- (B2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (C2) -- (D2) -- (E2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=red!30] (B2) -- (D2) -- (E2) -- cycle;
          \draw[thick, rounded corners=1pt, fill=green!30] (C3) -- (D3) -- (E3) -- cycle;
          \draw[thick, rounded corners=1pt, fill=green!30] (A3) -- (D3) -- (E3) -- cycle;
          \draw[thick, rounded corners=1pt, fill=green!30] (C3) -- (A3) -- (E3) -- cycle;
          \draw[thick, rounded corners=1pt, fill=green!30] (C3) -- (D3) -- (A3) -- cycle;

          % Edges
          \draw[dashed, line cap=round] (D1) -- (E1);
          \draw[dashed, line cap=round] (D2) -- (E2);
          \draw[dashed, line cap=round] (D3) -- (E3);
          % \draw[dashed, line cap=round] (C2) -- (B2);

          \draw (B1) node[anchor=north west] {B};
          \draw (C3) node[anchor=south west] {C};
          \draw (D1) node[anchor=south east] {D};
          \draw (E1) node[anchor=south west] {E};

        \end{tikzpicture}
